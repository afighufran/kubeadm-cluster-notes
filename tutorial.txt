# 1. bikin direktori keyrings dulu
/etc/apt/keyrings

# 2. tambahin key untuk k8s v1.36
curl -fsSL https://pkgs.k8s.io/core:/stable:/v1.36/deb/Release.key | sudo gpg --dearmor -o /etc/apt/keyrings/kubernetes-apt-keyring-v136.gpg

# 3. tambahin repo ke sources list
echo 'deb [signed-by=/etc/apt/keyrings/kubernetes-apt-keyring.gpg] https://pkgs.k8s.io/core:/stable:/v1.30/deb/ /' | sudo tee /etc/apt/sources.list.d/kubernetes.list

# 4. update, terus install komponen kubeadm nodenya
sudo apt-get update
sudo apt-get install -y kubelet kubeadm kubectl

# 5. kunci versi biar tidak keupgrade versinya
sudo apt-mark hold kubelet kubeadm kubectl

# 6. bikin cluster
sudo kubeadm init --pod-network-cidr=10.244.0.0/16 --apiserver-advertise-address=ip_control_plane

# 7. setup akses kubectl biar bisa di pake pada user biasa
mkdir -p $HOME/.kube
sudo cp -i /etc/kubernetes/admin.conf $HOME/.kube/config
sudo chown $(id -u):$(id -g) $HOME/.kube/config

# 8. install CNI flannel di control plane
kubectl apply -f https://github.com/flannel-io/flannel/releases/latest/download/kube-flannel.yml

# 9. join cluster
sudo kubeadm join your_ip:6443 --token <your_token> \
        --discovery-token-ca-cert-hash sha256:<your_hash>
